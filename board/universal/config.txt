# board/universal/config.txt
# Used by BR2_PACKAGE_RPI_FIRMWARE_CONFIG_FILE for per-board builds.
# The RPi bootloader reads this from the FAT boot partition.
#
# NOTE: for the universal multi-board image (merge-boot target), the
# Makefile copies this same file but the [piN] kernel= lines are what
# select the right kernel per board at runtime.
#
# Section order matters: lines in [all] apply first, then the matching
# [piN] section overrides kernel= for that SoC family.

[all]
arm_64bit=1
kernel=Image
dtoverlay=miniuart-bt
dtparam=i2c_arm=on
# Rack UI panel (I2C slave @ 0x36) is specified for 100 kHz; the OLED
# tolerates 400 kHz but the bus must respect the slowest device
dtparam=i2c_arm_baudrate=100000
enable_uart=1
dtparam=spi=on
dtoverlay=mu-cell-bb_raspberrypi

# Per-SoC-family kernel selection for the universal multi-board image.
# Per-board builds ignore these sections (they ship a single kernel=Image
# and genimage renames it, so [all] kernel=Image never matches there either).
[pi3]
kernel=kernel-rpi3.img

[pi4]
kernel=kernel-rpi4.img

[pi5]
kernel=kernel-rpi5.img
